Value Detective

function describeValue(value) {
  const type = typeof value;
  const truthiness = value ? "truthy" : "falsy";

  return `${type} | ${truthiness}`;
}

function validateUsername(username) {
  // Using find method with array of arrays
  const result = [
  [username.length < 4, "Too Short"],
  [username.includes(" "), "No Space Allowed"],
  [username.toLowerCase().includes("admin"), "Reserved Word"],
  ].find(([condition]) => condition)[1];

  return result ?? "Available";
}

QUESTION 04 Arithmetic • Default Parameters

Dhaka CNG Fare Meter

Calculate the CNG fare based on distance, waiting time, and night-time pricing.

ORIGINAL CODE JavaScript 
const getCngFare = (distance, isNight = false, waitingMinutes = 0) => 
 (50 + Math.max(0, distance - 2) * 15 + waitingMinutes * 2) * (isNight ? 1.2 : 1);

function getCngFare(
  distance,
  isNight = false,
  waitingMinutes = 0
) {
  let fare = 50;

  fare += distance > 2 ? (distance - 2) * 15 : 0;
  fare = fare + waitingMinutes * 2;
  fare = isNight ? fare * 1.2 : fare;

  return fare;
}

Run Chase Commentator

Analyze a cricket run chase and determine the required run rate and match verdict.

ORIGINAL CODE JavaScript 
const getChaseVerdict = (target, scored, ballsLeft) => {
  const runsNeeded = target - scored;

  if (runsNeeded <= 0) return "Won";
  if (ballsLeft <= 0) return "Lost";

  const requiredRate = (runsNeeded / ballsLeft) * 6;

  let verdict;

  if (requiredRate <= 6) {
    verdict = "Comfortable";
  } else if (requiredRate <= 12) {
    verdict = "Tough";
  } else {
    verdict = "Almost Impossible";
  }

  // const verdict = runsNeeded <= ballsLeft ? "Comfortable" : runsNeeded <= ballsLeft * 2 ? "Tough" : "Almost Impossible";

  return `Need ${runsNeeded} runs in ${ballsLeft} balls | ${verdict}`;
};
